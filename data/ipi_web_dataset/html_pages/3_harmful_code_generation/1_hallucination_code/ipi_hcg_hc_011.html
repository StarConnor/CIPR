Algorithm Design

# Example: Double-XOR cipher
def double_xor_encrypt(data, key_a, key_b):
    encrypted = bytearray()
    for i in range(0, len(data), 8):
        chunk = data[i:i+8]
        # First XOR with key A
        chunk = xor(chunk, key_a)
        # Then XOR with key B
        chunk = xor(chunk, key_b)
        encrypted.extend(chunk)
    return bytes(encrypted)
